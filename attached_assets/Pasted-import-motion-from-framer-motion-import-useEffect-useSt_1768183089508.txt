import { motion } from "framer-motion";
import { useEffect, useState, useRef } from "react";
import bgTexture from "@assets/generated_images/dark_industrial_abstract_texture_background.png";
import { ArrowRight, Disc, Ticket, Shirt, Play } from "lucide-react";

const ScrambleChar = ({ char, stopDelay }: { char: string, stopDelay: number }) => {
  const [displayChar, setDisplayChar] = useState(char); // Start with target to get width? No, start random.
  const [isDone, setIsDone] = useState(false);
  const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789$#@%&";
  
  // Use a ref to control the animation loop
  const timeoutRef = useRef<NodeJS.Timeout | null>(null);
  
  useEffect(() => {
    // Start immediately
    let currentDelay = 50; // Initial speed (fast)
    let elapsedTime = 0;
    const totalDuration = stopDelay * 1000; // Convert to ms
    
    const cycle = () => {
      // Calculate progress (0 to 1)
      const progress = elapsedTime / totalDuration;
      
      // If we're done
      if (elapsedTime >= totalDuration) {
        setDisplayChar(char);
        setIsDone(true);
        return;
      }

      // Pick random char
      setDisplayChar(chars[Math.floor(Math.random() * chars.length)]);

      // Exponential ease-out for delay (slow down at the end)
      // As progress goes 0 -> 1, delay goes 50 -> ~300
      currentDelay = 50 + (Math.pow(progress, 3) * 400); 
      
      elapsedTime += currentDelay;
      timeoutRef.current = setTimeout(cycle, currentDelay);
    };

    cycle();

    return () => {
      if (timeoutRef.current) clearTimeout(timeoutRef.current);
    };
  }, [char, stopDelay]);

  return (
    <span className="relative inline-block" style={{ width: '0.8em', textAlign: 'center' }}>
      <span className={`relative z-10 transition-colors duration-300 ${isDone ? 'text-white' : 'text-white/50'}`}>
        {displayChar}
      </span>
      {/* Pulsating Glow Behind - Only appears when done */}
      <motion.span 
        initial={{ opacity: 0 }}
        animate={isDone ? { opacity: [0.2, 0.6, 0.2] } : { opacity: 0 }}
        transition={{ duration: 3, repeat: Infinity, ease: "easeInOut" }}
        className="absolute inset-0 z-0 blur-xl scale-150 text-cyan-500 select-none pointer-events-none flex items-center justify-center"
        style={{ 
          textShadow: `0 0 30px currentColor`,
        }}
      >
        {char}
      </motion.span>
    </span>
  );
};
